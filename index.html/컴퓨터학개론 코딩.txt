import React, { useState } from 'react';
import { 
  AlertCircle, 
  CheckCircle, 
  BookOpen, 
  Layers, 
  Plus, 
  Minus, 
  GraduationCap, 
  ArrowRight,
  Info,
  Star,
  Flame,
  Calendar
} from 'lucide-react';

// 1. 실패한 필수 과목 데이터 (스크린샷 기준)
const FAILED_SUBJECTS = {
  major: [
    { id: 'cs', name: '컴퓨터학개론', credits: 3, type: '전공' },
    { id: 'calc', name: '미적분학', credits: 3, type: '전공' },
    { id: 'physics', name: '물리학개론', credits: 3, type: '전공' },
    { id: 'stats', name: '통계학개론', credits: 3, type: '전공' },
  ],
  essential: [
    { id: 'english', name: '대학영어', credits: 3, type: '일반 교육 필수' },
    { id: 'writing', name: '글쓰기', credits: 3, type: '일반 교육 필수' },
  ]
};

// 2. 각 과목별 대체 추천 과목 데이터 (에타 스타일 뱃지 포함)
const RECOMMENDATIONS = {
  cs: [
    { id: 'alt-cs-1', name: '파이썬 프로그래밍 기초', credits: 3, type: '핵심교양', rating: '4.8', passRate: '95%', reason: '컴퓨터학개론 대체 가능' },
    { id: 'alt-cs-2', name: '웹 테크놀로지 이해', credits: 3, type: '균형교양', rating: '4.3', passRate: '88%', reason: 'IT 기초 역량 충족' },
  ],
  calc: [
    { id: 'alt-calc-1', name: '생활 속의 수학', credits: 3, type: '균형교양', rating: '4.5', passRate: '92%', reason: '미적분학보다 부드러운 접근' },
  ],
  physics: [
    { id: 'alt-phys-1', name: '우주의 이해', credits: 3, type: '균형교양', rating: '4.2', passRate: '90%', reason: '과학적 사고 학점 대체' },
  ],
  stats: [
    { id: 'alt-stats-1', name: '빅데이터와 현대사회', credits: 3, type: '핵심교양', rating: '4.6', passRate: '94%', reason: '통계학 대안 과목' },
  ],
  english: [
    { id: 'alt-eng-1', name: '시사 영어 토론', credits: 3, type: '균형교양', rating: '4.7', passRate: '91%', reason: '대학영어 필수 학점 대체' },
  ],
  writing: [
    { id: 'alt-writ-1', name: '스토리텔링의 이해', credits: 3, type: '균형교양', rating: '4.4', passRate: '89%', reason: '글쓰기 학점 인정 가능' },
  ]
};

export default function CourseRegistrationSimulator() {
  // 현재 어떤 탭에 있는지 관리 ('diagnose', 'match', 'timetable')
  const [activeTab, setActiveTab] = useState('diagnose');
  // 체크한 실패 과목 ID 목록
  const [selectedFailures, setSelectedFailures] = useState<string[]>([]);
  // 최종 시간표 장바구니에 담은 대체 과목 목록
  const [cart, setCart] = useState<any[]>([]);

  const MAX_CREDITS = 18;
  const currentCredits = cart.reduce((sum, item) => sum + item.credits, 0);
  const progressPercentage = Math.min((currentCredits / MAX_CREDITS) * 100, 100);

  // 실패 과목 체크박스 토글
  const handleToggleFailure = (id: string) => {
    if (selectedFailures.includes(id)) {
      setSelectedFailures(selectedFailures.filter(item => item !== id));
      // 과목 해제 시 관련 추천 과목도 장바구니에서 자동 제거
      const relatedRecIds = RECOMMENDATIONS[id as keyof typeof RECOMMENDATIONS]?.map(r => r.id) || [];
      setCart(cart.filter(item => !relatedRecIds.includes(item.id)));
    } else {
      setSelectedFailures([...selectedFailures, id]);
    }
  };

  // 대체 과목 장바구니 담기/제거
  const handleToggleCart = (course: any) => {
    const isExist = cart.some(item => item.id === course.id);
    if (isExist) {
      setCart(cart.filter(item => item.id !== course.id));
    } else {
      if (currentCredits + course.credits > MAX_CREDITS) {
        alert(`최대 신청 학점(${MAX_CREDITS}학점)을 초과할 수 없습니다!`);
        return;
      }
      setCart([...cart, course]);
    }
  };

  // 선택된 실패 과목들에 매칭되는 모든 추천 과목 모으기
  const activeRecommendations = selectedFailures.flatMap(
    id => RECOMMENDATIONS[id as keyof typeof RECOMMENDATIONS] || []
  );

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 font-sans antialiased p-4 md:p-8">
      <div className="max-w-3xl mx-auto space-y-6">
        
        {/* Header */}
        <header className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
          <div>
            <h1 className="text-xl md:text-2xl font-bold text-blue-600 flex items-center gap-2 tracking-tight">
              <span className="p-2 bg-blue-50 rounded-xl text-blue-600">🎓</span>
              <div>
                <span>수강신청 구원 시뮬레이터</span>
                <p className="text-xs text-slate-400 font-normal mt-0.5">Save My Credits</p>
              </div>
            </h1>
          </div>
          <div className="bg-slate-100 text-slate-600 px-3 py-1.5 rounded-full text-xs font-semibold">
            2026학년도 1학기 새내기 플랜
          </div>
        </header>

        {/* 상단 고정 대시보드 (학점 현황 바) */}
        <section className="bg-white rounded-2xl p-5 shadow-sm border border-slate-100">
          <div className="flex justify-between items-center mb-3">
            <div className="flex items-center gap-2">
              <div className="p-1.5 bg-red-50 rounded-lg text-red-500">
                <AlertCircle size={16} />
              </div>
              <span className="text-sm font-semibold text-slate-700">현재 신청 학점</span>
            </div>
            <div className="flex items-center gap-1.5">
              <span className="text-xl font-black text-slate-900">{currentCredits}</span>
              <span className="text-slate-400 text-sm">/ {MAX_CREDITS} 학점</span>
              <span className={`text-[11px] px-2 py-0.5 rounded-full font-bold ${currentCredits === MAX_CREDITS ? 'bg-emerald-100 text-emerald-700' : 'bg-red-100 text-red-600'}`}>
                {currentCredits === MAX_CREDITS ? '완료' : '부족'}
              </span>
            </div>
          </div>
          
          <div className="w-full bg-slate-100 rounded-full h-3 overflow-hidden">
            <div 
              className="bg-blue-600 h-full rounded-full transition-all duration-500 ease-out"
              style={{ width: `${progressPercentage}%` }}
            ></div>
          </div>
          <div className="flex justify-between text-[10px] text-slate-400 mt-1.5">
            <span>0학점</span>
            <span>18학점</span>
          </div>
        </section>

        {/* 3단 탭 내비게이션 바 */}
        <nav className="grid grid-cols-3 bg-white p-1 rounded-xl shadow-sm border border-slate-100">
          <button 
            onClick={() => setActiveTab('diagnose')}
            className={`py-3 text-xs font-bold rounded-lg flex flex-col items-center gap-1 transition-all ${activeTab === 'diagnose' ? 'bg-blue-50 text-blue-600 shadow-sm' : 'text-slate-400 hover:text-slate-600'}`}
          >
            <AlertCircle size={16} />
            <span>실패 과목 진단</span>
          </button>
          <button 
            onClick={() => setActiveTab('match')}
            className={`py-3 text-xs font-bold rounded-lg flex flex-col items-center gap-1 transition-all ${activeTab === 'match' ? 'bg-blue-50 text-blue-600 shadow-sm' : 'text-slate-400 hover:text-slate-600'}`}
          >
            <BookOpen size={16} />
            <span>대안 과목 매칭</span>
          </button>
          <button 
            onClick={() => setActiveTab('timetable')}
            className={`py-3 text-xs font-bold rounded-lg flex flex-col items-center gap-1 transition-all ${activeTab === 'timetable' ? 'bg-blue-50 text-blue-600 shadow-sm' : 'text-slate-400 hover:text-slate-600'}`}
          >
            <Calendar size={16} />
            <span>나만의 시간표</span>
          </button>
        </nav>

        {/* 탭 내용 출력 영역 */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100 min-h-[300px]">
          
          {/* TAB 1: 실패 과목 진단 */}
          {activeTab === 'diagnose' && (
            <div className="space-y-5">
              <div className="flex items-start gap-2.5">
                <span className="text-xl">⚠️</span>
                <div>
                  <h3 className="font-bold text-slate-900 text-sm">어떤 필수 과목을 놓치셨나요?</h3>
                  <p className="text-xs text-slate-400 mt-0.5">실패한 과목을 선택해주세요</p>
                </div>
              </div>

              {/* 전공 기초 */}
              <div className="space-y-2">
                <h4 className="text-xs font-bold text-slate-400 flex items-center gap-1.5">
                  <Layers size={12} /> 전공 기초
                </h4>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                  {FAILED_SUBJECTS.major.map((sub) => {
                    const isChecked = selectedFailures.includes(sub.id);
                    return (
                      <label key={sub.id} className={`flex items-center justify-between p-4 rounded-xl border cursor-pointer transition-all ${isChecked ? 'border-blue-500 bg-blue-50/30' : 'border-slate-100 hover:bg-slate-50'}`}>
                        <div className="flex items-center gap-3">
                          <input type="checkbox" checked={isChecked} onChange={() => handleToggleFailure(sub.id)} className="rounded text-blue-600 border-slate-300 w-4 h-4" />
                          <div>
                            <p className="font-bold text-xs text-slate-800">{sub.name}</p>
                            <span className="text-[10px] text-slate-400">{sub.credits}학점</span>
                          </div>
                        </div>
                        <span className="text-[10px] bg-slate-100 text-slate-500 px-2 py-0.5 rounded-md font-medium">{sub.type}</span>
                      </label>
                    );
                  })}
                </div>
              </div>

              {/* 일반 교육 필수 */}
              <div className="space-y-2 pt-2">
                <h4 className="text-xs font-bold text-slate-400 flex items-center gap-1.5">
                  <GraduationCap size={12} /> 일반 교육 필수
                </h4>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                  {FAILED_SUBJECTS.essential.map((sub) => {
                    const isChecked = selectedFailures.includes(sub.id);
                    return (
                      <label key={sub.id} className={`flex items-center justify-between p-4 rounded-xl border cursor-pointer transition-all ${isChecked ? 'border-blue-500 bg-blue-50/30' : 'border-slate-100 hover:bg-slate-50'}`}>
                        <div className="flex items-center gap-3">
                          <input type="checkbox" checked={isChecked} onChange={() => handleToggleFailure(sub.id)} className="rounded text-blue-600 border-slate-300 w-4 h-4" />
                          <div>
                            <p className="font-bold text-xs text-slate-800">{sub.name}</p>
                            <span className="text-[10px] text-slate-400">{sub.credits}학점</span>
                          </div>
                        </div>
                        <span className="text-[10px] bg-slate-100 text-slate-500 px-2 py-0.5 rounded-md font-medium">{sub.type}</span>
                      </label>
                    );
                  })}
                </div>
              </div>
            </div>
          )}

          {/* TAB 2: 대안 과목 매칭 */}
          {activeTab === 'match' && (
            <div className="space-y-4">
              <div className="flex items-start gap-2.5">
                <span className="text-xl">💡</span>
                <div>
                  <h3 className="font-bold text-slate-900 text-sm">새내기를 위한 최적의 대안 교양</h3>
                  <p className="text-xs text-slate-400 mt-0.5">실패한 필수 과목을 채워줄 꿀교양 추천 목록입니다.</p>
                </div>
              </div>

              {selectedFailures.length === 0 ? (
                <div className="text-center py-12 text-slate-400 space-y-2">
                  <Info className="mx-auto" size={24} />
                  <p className="text-xs">먼저 첫 번째 탭에서 실패한 필수 과목을 체크해 주세요!</p>
                </div>
              ) : (
                <div className="space-y-2.5">
                  {activeRecommendations.map((course: any) => {
                    const isInCart = cart.some(item => item.id === course.id);
                    return (
                      <div key={course.id} className={`p-4 rounded-xl border transition-all flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 ${isInCart ? 'border-emerald-500 bg-emerald-50/10' : 'border-slate-100 bg-slate-50/50'}`}>
                        <div className="space-y-1">
                          <div className="flex items-center gap-2 flex-wrap">
                            <h4 className="font-bold text-xs text-slate-900">{course.name}</h4>
                            <span className="text-[9px] bg-blue-50 text-blue-600 font-bold px-1.5 py-0.2 rounded">
                              {course.type}
                            </span>
                            <span className="text-[9px] text-slate-400 bg-slate-100 px-1.5 py-0.2 rounded">
                              {course.reason}
                            </span>
                          </div>
                          <div className="flex items-center gap-3 text-[11px] text-slate-500">
                            <span>{course.credits}학점</span>
                            <span className="flex items-center gap-0.5 text-amber-500 font-medium">
                              <Star size={12} fill="currentColor" /> {course.rating}
                            </span>
                            <span className="flex items-center gap-0.5 text-red-500 font-medium">
                              <Flame size={12} /> 패스율 {course.passRate}
                            </span>
                          </div>
                        </div>
                        
                        <button
                          onClick={() => handleToggleCart(course)}
                          className={`w-full sm:w-auto px-3 py-1.5 rounded-lg text-xs font-bold shadow-sm transition-all ${isInCart ? 'bg-slate-200 text-slate-700 hover:bg-slate-300' : 'bg-blue-600 text-white hover:bg-blue-700'}`}
                        >
                          {isInCart ? '시간표에서 제거' : '내 시간표에 추가'}
                        </button>
                      </div>
                    );
                  })}
                </div>
              )}
            </div>
          )}

          {/* TAB 3: 나만의 시간표 */}
          {activeTab === 'timetable' && (
            <div className="space-y-4">
              <div className="flex items-start gap-2.5">
                <span className="text-xl">🗓️</span>
                <div>
                  <h3 className="font-bold text-slate-900 text-sm">최종 대안 수강 신청 시간표</h3>
                  <p className="text-xs text-slate-400 mt-0.5">구원 시뮬레이션을 통해 조립된 학점 대시보드입니다.</p>
                </div>
              </div>

              {cart.length === 0 ? (
                <div className="text-center py-12 text-slate-400 space-y-2">
                  <Info className="mx-auto" size={24} />
                  <p className="text-xs">아직 시간표에 담은 대체 과목이 없습니다.</p>
                </div>
              ) : (
                <div className="space-y-2">
                  <div className="bg-blue-50/50 p-3 rounded-xl text-xs text-blue-700 font-medium flex items-center gap-2">
                    <CheckCircle size={14} />
                    <span>안전하게 이번 학기 {currentCredits}학점을 방어했습니다!</span>
                  </div>
                  <div className="border border-slate-100 rounded-xl overflow-hidden">
                    <table className="w-full text-left border-collapse text-xs">
                      <thead>
                        <tr className="bg-slate-50 text-slate-400 font-semibold border-b border-slate-100">
                          <th className="p-3">과목명</th>
                          <th className="p-3">구분</th>
                          <th className="p-3 text-right">학점</th>
                        </tr>
                      </thead>
                      <tbody>
                        {cart.map((item) => (
                          <tr key={item.id} className="border-b border-slate-50 last:border-0 hover:bg-slate-50/40">
                            <td className="p-3 font-medium text-slate-800">{item.name}</td>
                            <td className="p-3 text-slate-500">{item.type}</td>
                            <td className="p-3 text-right font-bold text-blue-600">{item.credits}학점</td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>
              )}
            </div>
          )}

        </div>

      </div>
    </div>
  );
}
